function getOffsetDate(daysOffset) {
const d = new Date();
d.setDate(d.getDate() + daysOffset);
return d.toISOString().split('T')[0];
}
function parseDate(dateStr) {
if (!dateStr) return new Date();
const [y, m, d] = dateStr.split('-').map(Number);
return new Date(y, m - 1, d);
}
function formatDateISO(date) {
const y = date.getFullYear();
const m = String(date.getMonth() + 1).padStart(2, '0');
const d = String(date.getDate()).padStart(2, '0');
return `${y}-${m}-${d}`;
}
function formatDateGerman(dateStr) {
if (!dateStr) return '';
const d = parseDate(dateStr);
return d.toLocaleDateString('de-DE', { day: '2-digit', month: '2-digit', year: 'numeric' });
}
function getDaysDifference(startStr, endStr) {
const start = parseDate(startStr);
const end = parseDate(endStr);
const diffTime = end - start;
return Math.max(1, Math.round(diffTime / (1000 * 60 * 60 * 24)) + 1);
}
function getISOWeekInfo(date) {
const tempDate = new Date(date.valueOf());
const dayNum = tempDate.getDay() || 7;
tempDate.setDate(tempDate.getDate() + 4 - dayNum);
const yearStart = new Date(tempDate.getFullYear(), 0, 1);
const weekNo = Math.ceil((((tempDate - yearStart) / 86400000) + 1) / 7);
const monday = new Date(date.valueOf());
monday.setDate(monday.getDate() - (dayNum - 1));
const sunday = new Date(monday.valueOf());
sunday.setDate(sunday.getDate() + 6);
return { week: weekNo, year: tempDate.getFullYear(), monday, sunday };
}
function recalculateParentTaskDates(task) {

task.endDate = maxEnd;
}

return Math.round((doneCount / task.subtasks.length) * 100);
}

function calculateTimelineRange(tasks) {
let minDate = new Date();
let maxDate = new Date();
maxDate.setDate(maxDate.getDate() + 14);
if (tasks.length > 0) {
let minTime = Infinity;
let maxTime = -Infinity;
tasks.forEach(t => {
const start = parseDate(t.startDate).getTime();
const end = parseDate(t.endDate).getTime();
if (start < minTime) minTime = start;
if (end > maxTime) maxTime = end;
if (t.subtasks) {
t.subtasks.forEach(st => {
const sStart = parseDate(st.startDate).getTime();
const sEnd = parseDate(st.endDate).getTime();
if (sStart < minTime) minTime = sStart;
if (sEnd > maxTime) maxTime = sEnd;
});
}
});
if (minTime !== Infinity) minDate = new Date(minTime);
if (maxTime !== -Infinity) maxDate = new Date(maxTime);
}
if (appState.viewMode === 'weeks') {
const dayNum = minDate.getDay() || 7;
minDate.setDate(minDate.getDate() - (dayNum - 1));
}
minDate.setDate(minDate.getDate() - 7);
maxDate.setDate(maxDate.getDate() + 14);
return { minDate, maxDate };
}

const date = new Date(startDate.getTime() + i * dayMs);
const month = date.getMonth();
if (month !== currentMonth) {
currentMonth = month;
monthLabelElem = document.createElement('div');
monthLabelElem.className = 'border-r border-slate-200 dark:border-slate-700 px-2 flex items-center justify-center bg-slate-100/70 dark:bg-slate-800/70 truncate';
monthLabelElem.textContent = date.toLocaleDateString('de-DE', { month: 'long', year: 'numeric' });
topHeader.appendChild(monthLabelElem);
currentMonthCellSpan = 0;
}
currentMonthCellSpan++;
monthLabelElem.style.width = `${currentMonthCellSpan * cellWidth}px`;
const dayCell = document.createElement('div');
dayCell.style.width = `${cellWidth}px`;
dayCell.className = `flex-none border-r border-slate-100 dark:border-slate-800 flex items-center justify-center text-[10px] ${date.getDay() === 0 || date.getDay() === 6 ? 'bg-slate-100/50 dark:bg-slate-800/40 text-slate-400' : ''}`;
dayCell.textContent = date.getDate();
bottomHeader.appendChild(dayCell);
}
} else if (mode === 'weeks') {
let currentWeekKey = '';
let weekSpan = 0;
let weekLabelElem = null;
for (let i = 0; i < totalDays; i++) {
const date = new Date(startDate.getTime() + i * dayMs);
const weekInfo = getISOWeekInfo(date);
const weekKey = `${weekInfo.year}-W${weekInfo.week}`;
if (weekKey !== currentWeekKey) {
currentWeekKey = weekKey;
weekLabelElem = document.createElement('div');
weekLabelElem.className = 'border-r border-slate-300 dark:border-slate-600 px-2 flex items-center justify-center bg-brand-50/80 dark:bg-brand-950/40 text-brand-700 dark:text-brand-300 font-bold truncate text-[11px]';
const mStr = weekInfo.monday.toLocaleDateString('de-DE', { day: '2-digit', month: '2-digit' });
const sStr = weekInfo.sunday.toLocaleDateString('de-DE', { day: '2-digit', month: '2-digit' });
weekLabelElem.textContent = `KW ${weekInfo.week} • ${mStr} - ${sStr}`;
topHeader.appendChild(weekLabelElem);
weekSpan = 0;
}
weekSpan++;
weekLabelElem.style.width = `${weekSpan * cellWidth}px`;
const dayCell = document.createElement('div');
dayCell.style.width = `${cellWidth}px`;
const dayNames = ['S', 'M', 'D', 'M', 'D', 'F', 'S'];
const isMonday = date.getDay() === 1;
dayCell.className = `flex-none border-r ${isMonday ? 'border-slate-300 dark:border-slate-600 bg-slate-100 dark:bg-slate-800 font-bold' : 'border-slate-100 dark:border-slate-800/60'} flex items-center justify-center text-[10px] ${date.getDay() === 0 || date.getDay() === 6 ? 'text-slate-400' : ''}`;
dayCell.textContent = dayNames[date.getDay()];
bottomHeader.appendChild(dayCell);
}
} else if (mode === 'months') {
let currentYear = -1;
let currentYearCellSpan = 0;
let yearLabelElem = null;
for (let i = 0; i < totalDays; i++) {
const date = new Date(startDate.getTime() + i * dayMs);
const year = date.getFullYear();
if (year !== currentYear) {
currentYear = year;
yearLabelElem = document.createElement('div');
yearLabelElem.className = 'border-r border-slate-200 dark:border-slate-700 px-2 flex items-center justify-center bg-slate-100/70 dark:bg-slate-800/70';
yearLabelElem.textContent = year;
topHeader.appendChild(yearLabelElem);
currentYearCellSpan = 0;
}
currentYearCellSpan++;
yearLabelElem.style.width = `${currentYearCellSpan * cellWidth}px`;
const dayCell = document.createElement('div');
dayCell.style.width = `${cellWidth}px`;
dayCell.className = 'flex-none border-r border-slate-100 dark:border-slate-800 flex items-center justify-center text-[9px]';
if (date.getDate() === 1) {
dayCell.textContent = date.toLocaleDateString('de-DE', { month: 'narrow' });
dayCell.classList.add('font-bold', 'text-brand-600');
}
bottomHeader.appendChild(dayCell);
}
}

function attachGanttDragListeners(timelineMinDate, cellWidth) {
const bars = document.querySelectorAll('.gantt-bar');
const tooltip = document.getElementById('gantt-tooltip');
const tooltipTitle = document.getElementById('tooltip-title');
const tooltipDates = document.getElementById('tooltip-dates');
const tooltipDuration = document.getElementById('tooltip-duration');
bars.forEach(bar => {
bar.addEventListener('mousedown', (e) => {
e.preventDefault();
const taskId = bar.dataset.taskId;
const isSubtask = bar.dataset.isSubtask === 'true';
const subtaskId = bar.dataset.subtaskId;
const isResizeLeft = e.target.classList.contains('left');
const isResizeRight = e.target.classList.contains('right');
const isMove = !isResizeLeft && !isResizeRight;
const task = appState.tasks.find(t => t.id === taskId);
if (!task) return;
let itemObj = isSubtask ? task.subtasks.find(s => s.id === subtaskId) : task;
if (!itemObj) return;
const startX = e.clientX;
const initialLeft = parseFloat(bar.style.left) || 0;
const initialWidth = parseFloat(bar.style.width) || 0;
let newStartDateStr = itemObj.startDate || task.startDate;
let newEndDateStr = itemObj.endDate || task.endDate;
tooltipTitle.textContent = (isSubtask ? 'Unteraufgabe: ' : 'Aufgabe: ') + itemObj.title;
tooltip.classList.remove('hidden');
function onMouseMove(moveEvent) {
const deltaX = moveEvent.clientX - startX;
const deltaDays = Math.round(deltaX / cellWidth);
if (isMove) {
const initialStart = parseDate(itemObj.startDate || task.startDate);
const initialEnd = parseDate(itemObj.endDate || task.endDate);
const newStart = new Date(initialStart.getTime() + deltaDays * 86400000);
const newEnd = new Date(initialEnd.getTime() + deltaDays * 86400000);
newStartDateStr = formatDateISO(newStart);
newEndDateStr = formatDateISO(newEnd);
bar.style.left = `${initialLeft + (deltaDays * cellWidth)}px`;
} else if (isResizeRight) {
const durationDays = getDaysDifference(itemObj.startDate || task.startDate, itemObj.endDate || task.endDate);
const newDuration = Math.max(1, durationDays + deltaDays);
const initialStart = parseDate(itemObj.startDate || task.startDate);
const newEnd = new Date(initialStart.getTime() + (newDuration - 1) * 86400000);
newEndDateStr = formatDateISO(newEnd);
bar.style.width = `${newDuration * cellWidth}px`;
} else if (isResizeLeft) {
const initialStart = parseDate(itemObj.startDate || task.startDate);
const initialEnd = parseDate(itemObj.endDate || task.endDate);
const newStart = new Date(initialStart.getTime() + deltaDays * 86400000);
if (newStart <= initialEnd) {
newStartDateStr = formatDateISO(newStart);
bar.style.left = `${initialLeft + (deltaDays * cellWidth)}px`;
bar.style.width = `${initialWidth - (deltaDays * cellWidth)}px`;
}
}
tooltip.style.left = `${moveEvent.clientX + 12}px`;
tooltip.style.top = `${moveEvent.clientY + 12}px`;
const duration = getDaysDifference(newStartDateStr, newEndDateStr);
const kwInfo = getISOWeekInfo(parseDate(newStartDateStr));
tooltipDates.textContent = `${formatDateGerman(newStartDateStr)} - ${formatDateGerman(newEndDateStr)} (KW ${kwInfo.week})`;
tooltipDuration.textContent = `Dauer: ${duration} Tag(e)`;
}
function onMouseUp() {
document.removeEventListener('mousemove', onMouseMove);
document.removeEventListener('mouseup', onMouseUp);
tooltip.classList.add('hidden');
itemObj.startDate = newStartDateStr;
itemObj.endDate = newEndDateStr;
if (isSubtask) {
recalculateParentTaskDates(task);
}
saveState();
renderAll();
}
document.addEventListener('mousemove', onMouseMove);
document.addEventListener('mouseup', onMouseUp);
});
});
}

appState.collapsedTasks.add(taskId);

renderAll();
}
